Public Enum Materia {
  MATEMATICAS, LENGUAJE, HISTORIA, QUIMICA, INGLES, FISICA
}

Abstract Class PerfilBasico {
  - id: String
  - nombre: String
  - email: String
  
  + PerfilBasica(nombre, email) : Void
  + getId(): String
  + getNombre(): String
  + setNombre() : Void
  + getEmail(): String
  + setEmail() : Void
}

Class Estudiante extends PerfilBasico {
  - inscritas: List<Clase>

  + Estudiante(nombre, email) : Void (Id debe hacerse con UUID)
  + solicitarClase(clase: Clase): boolean
  + cancelarClase(clase: Clase): boolean
  + getInscritas(): List<Clase>
}

Class Tutor extends PerfilBasico {
  - materiasImpartidas: List<Materia>
  - clasesOfrecidas: List<Clase>
  - disponibilidad: List<Bloques>

  +Tutor(Nombre, email)
  + agregarMateria(materia: Materia): void
  + removerMateria(materia: Materia): void
  + crearClase(materia: Materia, tarifa: int, cuposMax: int): Clase
  + cancelarClase(clase: Clase): void
}

Class Clase {
  - idClase: String
  - tutor: Tutor
  - materia: Materia
  - tarifa: int
  - cuposMaximos: int
  - alumnosInscritos: List<Estudiante>
  - bloquesHorarios: List<BloqueHorario>

  + agregarBloque(bloque: BloqueHorario): boolean
  + inscribirEstudiante(estudiante: Estudiante): boolean
  + removerEstudiante(estudiante: Estudiante): boolean
  + hayConflictoHorario(otroBloque: BloqueHorario): boolean
  + getCuposDisponibles(): int
}

Class BloqueHorario {
  - fecha: LocalDate
  - horaInicio: LocalTime
  - horaFin: LocalTime 
  
  + coincideCon(otro: BloqueHorario): boolean
}
